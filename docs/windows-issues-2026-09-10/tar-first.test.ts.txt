const outputDir = process.env.ARCHON_TAR_PROBE_DIR;
if (!outputDir) throw new Error('ARCHON_TAR_PROBE_DIR is required');
const eventPath = join(outputDir, 'parent.jsonl');
const started = performance.now();
const trace = (event: string, fields: Record<string, unknown> = {}) => {
  appendFileSync(eventPath, `${JSON.stringify({ event, utc: new Date().toISOString(), elapsedMs: performance.now() - started, parentPid: process.pid, ...fields })}\n`);
};

// Copied from serve.test.ts at investigation baseline; no tar process builds this fixture.
function writeField(header: Uint8Array, offset: number, value: string, width: number): void {
  header.set(new TextEncoder().encode(value).subarray(0, width), offset);
}
function writeOctalField(header: Uint8Array, offset: number, value: number, width: number): void {
  writeField(header, offset, value.toString(8).padStart(width - 1, '0'), width - 1);
}
function tarHeader(name: string, size: number, typeflag: '0' | '5', mode: number): Uint8Array {
  const header = new Uint8Array(512);
  writeField(header, 0, name, 100);
  writeOctalField(header, 100, mode, 8);
  writeOctalField(header, 108, 0, 8);
  writeOctalField(header, 116, 0, 8);
  writeOctalField(header, 124, size, 12);
  writeOctalField(header, 136, 0, 12);
  header.fill(0x20, 148, 156);
  header[156] = typeflag.charCodeAt(0);
  writeField(header, 257, 'ustar', 6);
  writeField(header, 263, '00', 2);
  let checksum = 0;
  for (const byte of header) checksum += byte;
  writeField(header, 148, checksum.toString(8).padStart(6, '0'), 6);
  header[154] = 0x00;
  header[155] = 0x20;
  return header;
}
function concatBytes(blocks: Uint8Array[]): Uint8Array<ArrayBuffer> {
  const total = blocks.reduce((sum, block) => sum + block.length, 0);
  const out = new Uint8Array(total);
  let offset = 0;
  for (const block of blocks) { out.set(block, offset); offset += block.length; }
  return out;
}
const fixtureContent = '<html>ok</html>';
function buildWebTarball(indexHtml: string): Uint8Array<ArrayBuffer> {
  const body = new TextEncoder().encode(indexHtml);
  const padding = new Uint8Array((512 - (body.length % 512)) % 512);
  return new Uint8Array(Bun.gzipSync(concatBytes([
    tarHeader('web/', 0, '5', 0o755),
    tarHeader('web/index.html', body.length, '0', 0o644),
    body, padding, new Uint8Array(1024),
  ])));
}

it('observes the first real tar extraction with the normal 5000ms test budget', async () => {
  const bytes = buildWebTarball(fixtureContent);
  const hash = new Bun.CryptoHasher('sha256').update(bytes).digest('hex');
  const targetDir = join(outputDir, 'target');
  trace('probe_started', { bunVersion: Bun.version, bunRevision: Bun.revision, execPath: process.execPath, bytes: bytes.byteLength, sha256: hash, targetDir, testBudgetMs: 5000, productionBudgetMs: 60000 });
  const fetchSpy = spyOn(globalThis, 'fetch').mockImplementation(async () => new Response(bytes));
  const realSpawn = Bun.spawn.bind(Bun);
  let tarProc: Bun.Subprocess | undefined;
  let spawnCount = 0;
  let heartbeatCount = 0;
  const heartbeat = setInterval(() => trace('parent_heartbeat', { heartbeatCount: ++heartbeatCount, tarPid: tarProc?.pid, exitCode: tarProc?.exitCode, signalCode: tarProc?.signalCode }), 100);
  const spawnSpy = spyOn(Bun, 'spawn').mockImplementation(((command: string[], options: Parameters<typeof Bun.spawn>[1]) => {
    const spawnStart = performance.now();
    trace('spawn_enter', { spawnCount: ++spawnCount, command, stdinIsBlob: options?.stdin instanceof Blob, stdout: options?.stdout ?? 'default-pipe', stderr: options?.stderr });
    const proc = realSpawn(command, options);
    tarProc = proc as Bun.Subprocess;
    trace('spawn_returned', { tarPid: proc.pid, spawnMs: performance.now() - spawnStart });
    void proc.exited.then(exitCode => trace('observed_proc_exited', { tarPid: proc.pid, exitCode, signalCode: proc.signalCode }));
    return proc;
  }) as typeof Bun.spawn);
  try {
    await downloadWebDist('9.9.9', targetDir, hash);
    expect(readFileSync(join(targetDir, 'index.html'), 'utf8')).toBe(fixtureContent);
    expect(existsSync(`${targetDir}.tmp.tar.gz`)).toBe(false);
    trace('extraction_verified', { spawnCount });
  } catch (error) {
    trace('extraction_error', { error: error instanceof Error ? error.message : String(error) });
    throw error;
  } finally {
    clearInterval(heartbeat);
    spawnSpy.mockRestore();
    fetchSpy.mockRestore();
    trace('probe_finally');
  }
});
